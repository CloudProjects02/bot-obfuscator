  /* OKLCH accent tween.
  
     CSS cannot give us a smooth accent swap without washing out: a transition on
     any of the accent vars interpolates in sRGB, and the straight line between two
     saturated hues passes through the desaturated middle - measured at 1% chroma
     for orange->blue, a near-white grey. Measured alternatives: a <color> written
     in oklch() still interpolates in sRGB (17%), and color-mix over a transitioned
     var is no better (10%). Only rotating the HUE keeps chroma up (54%).
  
     So the tween runs here instead: interpolate in OKLCH (shortest hue arc, chroma
     and lightness eased independently) and write the SAME variable formats the CSS
     already expects - #rrggbb for --accent/--accent-2, "r, g, b" for --accent-hi
     and --accent-rgb. Nothing downstream changes; all ~268 existing var() uses
     keep working untouched.
  
     Element-level transitions stay suppressed for the duration (html.accent-swapping)
     so nothing lags behind or re-interpolates the frames in sRGB. */
  (function () {
    var VARS = ['--accent', '--accent-hi', '--accent-2', '--accent-rgb'];
    var LIST = { '--accent-hi': 1, '--accent-rgb': 1 };   // these are bare "r, g, b" lists
  
    function srgbToLin(c) { c /= 255; return c <= 0.04045 ? c / 12.92 : Math.pow((c + 0.055) / 1.055, 2.4); }
    function linToSrgb(c) { var v = c <= 0.0031308 ? c * 12.92 : 1.055 * Math.pow(c, 1 / 2.4) - 0.055;
      return Math.max(0, Math.min(255, Math.round(v * 255))); }
  
    function rgbToOklch(r, g, b) {
      var lr = srgbToLin(r), lg = srgbToLin(g), lb = srgbToLin(b);
      var l = Math.cbrt(0.4122214708 * lr + 0.5363325363 * lg + 0.0514459929 * lb);
      var m = Math.cbrt(0.2119034982 * lr + 0.6806995451 * lg + 0.1073969566 * lb);
      var s = Math.cbrt(0.0883024619 * lr + 0.2817188376 * lg + 0.6299787005 * lb);

      var A = 1.9779984951 * l - 2.4285922050 * m + 0.4505937099 * s;
      var B = 0.0259040371 * l + 0.7827717662 * m - 0.8086757660 * s;
      var C = Math.sqrt(A * A + B * B);
      var H = Math.atan2(B, A) * 180 / Math.PI; if (H < 0) H += 360;
      return [L, C, H];
    }

      var h = H * Math.PI / 180, A = Math.cos(h) * C, B = Math.sin(h) * C;
      var l = L + 0.3963377774 * A + 0.2158037573 * B;
      var m = L - 0.1055613458 * A - 0.0638541728 * B;
      var s = L - 0.0894841775 * A - 1.2914855480 * B;

        -1.2684380046 * l + 2.6097574011 * m - 0.3413193965 * s,
        -0.0041960863 * l - 0.7034186147 * m + 1.7076147010 * s
      ];
    }
    function inGamut(lin) {
      for (var i = 0; i < 3; i++) if (lin[i] < -0.0001 || lin[i] > 1.0001) return false;
      return true;
    }
    /* Gamut-map by REDUCING CHROMA, not by clamping channels.
       The accent tints sit near L=85%, where a vivid intermediate hue can fall
       outside sRGB. Clamping each channel independently pushes such a colour
       toward white - which put a residual pale flash back on .pc-key-ic. Binary-
       searching the largest in-gamut chroma keeps the hue and lightness and simply
       makes the step slightly less vivid, which is imperceptible. */
    function oklchToRgb(L, C, H) {
      var lin = oklchRaw(L, C, H);
      if (!inGamut(lin)) {
        var lo = 0, hi = C;
        for (var i = 0; i < 12; i++) {
          var mid = (lo + hi) / 2;
          if (inGamut(oklchRaw(L, mid, H))) lo = mid; else hi = mid;
        }
        lin = oklchRaw(L, lo, H);
      }
      return [linToSrgb(lin[0]), linToSrgb(lin[1]), linToSrgb(lin[2])];

      if (m) { var n = parseInt(m[1], 16); return [(n >> 16) & 255, (n >> 8) & 255, n & 255]; }
      var nums = v.match(/-?\d*\.?\d+/g);
      return nums && nums.length >= 3 ? [ +nums[0], +nums[1], +nums[2] ] : null;
    }

        lch[v] = { a: a, b: b, dh: dh };
      });

      var ease = function (t) { return t < 0.5 ? 2 * t * t : 1 - Math.pow(-2 * t + 2, 2) / 2; };
  
      var finish = function () {
        usable.forEach(function (v) { root.style.removeProperty(v); });   // hand back to the stylesheet
        root.classList.remove('accent-swapping');
      };

        });

        else { if (live === self) live = null; finish(); }
      })(t0);
    };

      function land(lat,lng){
        var px=((lng+180)/360*MW)|0, py=((90-lat)/180*MH)|0;
        if(px<0)px=0; else if(px>=MW)px=MW-1;
